
import streamlit as st
import requests
import numpy as np
import faiss

from io import BytesIO
from pypdf import PdfReader
from sentence_transformers import SentenceTransformer
from langchain_text_splitters import RecursiveCharacterTextSplitter


# Page configuration
st.set_page_config(
    page_title="AI Document Assistant",
    page_icon="📚",
    layout="wide"
)

st.title("📚 AI Document Assistant")
st.write("Upload a PDF and ask questions about its contents.")


# Load embedding model and process PDF
@st.cache_resource
def load_rag_system(pdf_bytes):

    reader = PdfReader(BytesIO(pdf_bytes))

    all_pages = []

    for page_number, page in enumerate(reader.pages, start=1):
        page_text = page.extract_text()

        if page_text:
            all_pages.append({
                "page": page_number,
                "text": page_text
            })

    splitter = RecursiveCharacterTextSplitter(
        chunk_size=1000,
        chunk_overlap=200
    )

    chunks = []

    for page in all_pages:
        page_chunks = splitter.split_text(page["text"])

        for chunk in page_chunks:
            chunks.append({
                "page": page["page"],
                "text": chunk
            })

    model = SentenceTransformer("all-MiniLM-L6-v2")

    texts = [chunk["text"] for chunk in chunks]

    embeddings_list = []

    for i in range(0, len(texts), 50):
        batch = texts[i:i+50]

        batch_embeddings = model.encode(
            batch,
            normalize_embeddings=True
        )

        embeddings_list.extend(batch_embeddings)

    embeddings = np.array(
        embeddings_list,
        dtype="float32"
    )

    index = faiss.IndexFlatIP(embeddings.shape[1])
    index.add(embeddings)

    return model, chunks, index


# Hybrid search
def hybrid_search(query, model, chunks, index, top_k=5):

    query_embedding = model.encode(
        [query],
        normalize_embeddings=True
    ).astype("float32")

    scores, indices = index.search(query_embedding, 10)

    selected = {}

    # Semantic results
    for rank, idx in enumerate(indices[0]):
        if idx != -1:
            selected[int(idx)] = 10 - rank

    # Keyword results
    keywords = query.lower().split()

    keyword_scores = []

    for i, chunk in enumerate(chunks):
        text = chunk["text"].lower()

        score = sum(
            1 for word in keywords
            if len(word) > 2 and word in text
        )

        if score > 0:
            keyword_scores.append((score, i))

    keyword_scores.sort(reverse=True)

    for rank, (score, idx) in enumerate(keyword_scores[:10]):
        selected[idx] = selected.get(idx, 0) + score + 5

    ranked = sorted(
        selected.items(),
        key=lambda x: x[1],
        reverse=True
    )

    return [chunks[idx] for idx, score in ranked[:top_k]]


# PDF uploader
uploaded_file = st.file_uploader(
    "Upload your PDF Document",
    type=["pdf"]
)


if uploaded_file is not None:

    pdf_bytes = uploaded_file.getvalue()

    st.success(f"Uploaded: {uploaded_file.name}")

    with st.spinner("Reading PDF and preparing AI search system..."):

        try:
            model, chunks, index = load_rag_system(pdf_bytes)

            st.success(
                f"Document processed successfully! "
                f"{len(chunks)} text chunks created."
            )

        except Exception as e:
            st.error(f"PDF processing error: {e}")
            st.stop()

    # Question input
    question = st.text_input(
        "Enter your Question:",
        placeholder="What is the eligibility criteria?"
    )

    if st.button("Ask AI"):

        if not question.strip():
            st.warning("Please enter a question.")

        else:

            with st.spinner("AI is finding your answer..."):

                try:
                    results = hybrid_search(
                        question,
                        model,
                        chunks,
                        index
                    )

                    context = ""

                    for result in results:
                        context += (
                            f"\nPage {result['page']}:\n"
                            f"{result['text']}\n"
                        )

                    prompt = f"""
You are a university document assistant.

Answer using ONLY the provided document context.

Instructions:
1. Give the actual answer.
2. Include all relevant conditions.
3. Use simple language.
4. Mention the source page numbers.
5. Do not invent information.
6. If information is missing, say so.

Document Context:
{context}

Question:
{question}

Answer:
"""

                    response = requests.post(
                        "http://localhost:11434/api/generate",
                        json={
                            "model": "llama3.2:1b",
                            "prompt": prompt,
                            "stream": False,
                            "options": {
                                "temperature": 0,
                                "num_predict": 300
                            }
                        },
                        timeout=180
                    )

                    response.raise_for_status()

                    answer = response.json()["response"]

                    st.subheader("🤖 AI Answer")
                    st.write(answer)

                    st.subheader("📄 Source Documents")

                    for result in results:
                        st.write(f"Page {result['page']}")
                        st.caption(result["text"])

                except Exception as e:
                    st.error(f"Something went wrong: {e}")

else:
    st.info("Please upload a PDF document to get started.")
